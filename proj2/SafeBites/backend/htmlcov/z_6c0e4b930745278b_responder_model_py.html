1""" 

2Defines Pydantic models for handling dish and menu-related information 

3in the SafeBites backend. 

4 

5Models: 

6 

7- AllergenInfo: Details about potential allergens in a dish. 

8- NutritionFacts: Nutritional information for a dish. 

9- DishResult: Represents a dish retrieved from the database or service. 

10- InfoResult: Contains requested information about a specific dish. 

11- QueryResponse: Wraps a single query and its corresponding result(s). 

12- FinalResponse: Aggregates all query responses for a user session. 

13 

14These models standardize the data passed between services and responses 

15returned to clients in the chat and menu system. 

16""" 

17from typing import List, Optional, Dict, Any, Union 

18from pydantic import BaseModel, Field 

19from datetime import datetime 

20 

21class AllergenInfo(BaseModel): 

22 """ 

23 Represents detailed information about an allergen detected or specified in a dish. 

24 

25 Attributes: 

26 allergen (str): The name of the allergen (e.g., "peanuts", "gluten"). 

27 confidence (float): Confidence score (0–1) indicating the certainty of allergen presence. 

28 why (Optional[str]): Explanation or reasoning behind allergen detection (e.g., "ingredient inference"). 

29 """ 

30 allergen:str 

31 confidence:float 

32 why:Optional[str] 

33 

34class NutritionFacts(BaseModel): 

35 """ 

36 Represents nutritional details of a dish, including macronutrient and caloric breakdown. 

37 

38 Attributes: 

39 calories (Optional[Dict[str, Any]]): Information about calorie content. 

40 protein (Optional[Dict[str, Any]]): Protein content details. 

41 fat (Optional[Dict[str, Any]]): Fat content details. 

42 carbohydrates (Optional[Dict[str, Any]]): Carbohydrate content details. 

43 sugar (Optional[Dict[str, Any]]): Sugar content details. 

44 fiber (Optional[Dict[str, Any]]): Fiber content details. 

45 """ 

46 calories: Optional[Dict[str,Any]] = None 

47 protein : Optional[Dict[str,Any]] = None 

48 fat : Optional[Dict[str,Any]] = None 

49 carbohydrates : Optional[Dict[str,Any]] = None 

50 sugar : Optional[Dict[str,Any]] = None 

51 fiber : Optional[Dict[str,Any]] = None

53class CompatibilityScoreBreakdown(BaseModel): 

54 """ 

55 Detailed breakdown of AI compatibility score for a dish. 

56 

57 Attributes: 

58 overall_score (int): Overall compatibility score 0-100. 

59 allergen_safety (Dict[str, Any]): Allergen safety analysis. 

60 nutrition_match (Dict[str, Any]): Nutrition matching with health goals. 

61 taste_preference (Dict[str, Any]): Taste and cuisine preference match. 

62 dietary_pattern (Dict[str, Any]): Dietary pattern alignment. 

63 recommendation (str): AI-generated recommendation. 

64 alternative_suggestions (List[Dict[str, Any]]): Better alternatives if score is low. 

65 """ 

66 overall_score: int 

67 allergen_safety: Dict[str, Any] 

68 nutrition_match: Dict[str, Any] 

69 taste_preference: Dict[str, Any] 

70 dietary_pattern: Dict[str, Any] 

71 recommendation: str 

72 alternative_suggestions: List[Dict[str, Any]] = []

78 Attributes:

80 restaurant_id (str): ID of the restaurant offering the dish. 

81 name (str): Name of the dish. 

82 description (str): Description of the dish. 

83 price (Optional[float]): Price of the dish. 

84 ingredients (Optional[List[str]]): List of ingredients used.

88 serving_size (Optional[str]): The serving size or portion information.

93 restaurant_id:str

141class FinalResponse(BaseModel): 

142 """ 

143 Represents the complete structured response for a user session, 

144 combining all processed queries and their results. 

145 

146 Attributes: 

147 user_id (str): Unique identifier for the user. 

148 session_id (str): Unique session identifier. 

149 restaurant_id (str): ID of the restaurant context for the conversation. 

150 original_query (str): The original user input. 

151 responses (List[QueryResponse]): A list of structured query responses. 

152 status (str): Processing status (e.g., 'pending', 'completed', 'failed'). 

153 timestamp (str): ISO-formatted timestamp indicating when the response was generated. 

154 """ 

155 user_id:str 

156 session_id:str 

157 restaurant_id:str 

158 original_query:str 

159 responses : List[QueryResponse] 

160 status : str 

161 timestamp : str = Field(default_factory=lambda: datetime.utcnow().isoformat())
