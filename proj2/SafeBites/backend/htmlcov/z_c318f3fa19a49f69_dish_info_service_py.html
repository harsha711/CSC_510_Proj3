29def derive_dish_info_intent(query): 

30 """ 

31 Determine whether a user query requires restaurant-specific menu data or general food knowledge. 

32 

33 This function uses an LLM to analyze the query and classify it as either: 

34 - "requires_menu_data" (when query involves dishes, ingredients, allergens, or calories) 

35 - "general_knowledge" (when query is conceptual or unrelated to a restaurant's menu, 

36 OR when the context already contains sufficient information to answer) 

37 

38 Args: 

39 query (str): The user's natural language question (may include context). 

40 

41 Returns: 

42 dict: JSON-like dictionary containing: 

43 - type (str): Either "requires_menu_data" or "general_knowledge". 

44 - message (Optional[str]): Error message if parsing fails. 

45 """ 

46 logging.debug(f"Deriving intent for query: {query}") 

47 

48 prompt = ChatPromptTemplate.from_template(""" 

49 You are an intent analyzer for a food assistant. 

50 

51 Given a query (which may include additional context), decide whether you need to fetch restaurant menu data or if you can answer from the provided context. 

52 

53 Possible outputs: 

54 - "requires_menu_data" → if the question is about dishes, ingredients, allergens, or calories AND the context does not contain the answer. 

55 - "general_knowledge" → if (1) the question is conceptual/general, OR (2) the "Additional context" section already contains sufficient information to answer the query. 

56 

57 **CRITICAL**: If the query includes an "Additional context:" section that contains the answer to the user's question, return "general_knowledge". 

58 

59 Query: {query} 

60 

61 CRITICAL: Your response must ONLY be valid JSON. Do not include any explanation, markdown formatting, or additional text. 

62 

63 Output format (JSON only): 

64 {{"type": "requires_menu_data"}} OR {{"type": "general_knowledge"}} 

65 

66 Remember: Output ONLY the JSON object, nothing else. 

67 """) 

68 response = llm.invoke(prompt.format_messages(query=query)) 

69 logging.debug(f"LLM Intent Response: {response.content}") 

70 try: 

71 # Check if response is empty 

72 if not response.content or not response.content.strip(): 

73 logger.warning(f"Empty LLM response for derive_dish_info_intent query: {query}. Defaulting to general_knowledge.") 

74 return IntentResponse(type="general_knowledge") 

75 

76 content = response.content.strip() 

77 

78 # Try to extract JSON from markdown code blocks if present 

79 if content.startswith("```"): 

80 content = content.replace("```json", "").replace("```", "").strip() 

81 

82 intent_json = json.loads(content) 

83 return IntentResponse(**intent_json) 

84 except json.JSONDecodeError as e: 

85 logger.error(f"JSON decode error in derive_dish_info_intent. Response: {response.content[:500]}. Error: {str(e)}") 

86 logger.warning("Falling back to general_knowledge due to JSON parse error") 

87 return IntentResponse(type="general_knowledge") 

88 except Exception as e: 

89 logging.error(f"Error in derive_dish_info_intent: {str(e)}") 

90 raise GenericException(f"Unexpected error: {str(e)}")

93def handle_general_knowledge(query): 

94 """ 

95 Handle general food knowledge queries. 

96 

97 Args: 

98 query (str): User query text. 

99 

100 Returns: 

101 GeneralKnowledgeResponse: Pydantic model with answer text. 

102 """ 

103 logging.debug(f"Handling general knowledge query: {query}") 

104 prompt = ChatPromptTemplate.from_template(""" 

105 You are a food assistant. Answer the following query using general food knowledge only. 

106 Do NOT assume restaurant-specific information unless explicitly mentioned. 

107 Query: {query} 

108 

109 CRITICAL: Your response must ONLY be valid JSON. Do not include any explanation, markdown formatting, or additional text. 

110 

111 Output format (JSON only): 

112 {{"answer": "your answer to the query"}} 

113 

114 Remember: Output ONLY the JSON object, nothing else. 

115 """) 

116 response = llm.invoke(prompt.format_messages(query=query)) 

117 logging.debug(f"LLM Response: {response.content}") 

118 try: 

119 # Check if response is empty 

120 if not response.content or not response.content.strip(): 

121 logger.warning(f"Empty LLM response for general knowledge query: {query}. Returning error message.") 

122 return GeneralKnowledgeResponse(answer="I apologize, but I couldn't generate a response. Please try again.") 

123 

124 content = response.content.strip() 

125 

126 # Try to extract JSON from markdown code blocks if present 

127 if content.startswith("```"): 

128 content = content.replace("```json", "").replace("```", "").strip() 

129 

130 answer_json = json.loads(content) 

131 return GeneralKnowledgeResponse(**answer_json) 

132 except json.JSONDecodeError as e: 

133 logger.error(f"JSON decode error in handle_general_knowledge. Response: {response.content[:500]}. Error: {str(e)}") 

134 return GeneralKnowledgeResponse(answer=f"I understand you're asking: {query}. However, I encountered an error processing the response. Please try rephrasing your question.") 

135 except Exception as e: 

136 logger.error(f"Error in handle_general_knowledge: {str(e)}") 

137 raise GenericException(str(e))

139def handle_food_item_query(query, restaurant_id=None): 

140 """ 

141 Retrieve and structure dish data from the semantic search. 

142 

143 Args: 

144 query (str): User query text. 

145 restaurant_id (Optional[str]): Restaurant identifier. 

146 

147 Returns: 

148 List[DishData]: List of structured dish data. 

149 

150 Raises: 

151 NotFoundException: If no dishes are found. 

152 """ 

153 logging.debug(f"Handling food item query: {query} for restaurant_id: {restaurant_id}") 

154 hits = semantic_retrieve_with_negation(query, restaurant_id=restaurant_id) 

155 

156 if not hits: 

157 raise NotFoundException(f"No relevant dishes found for query {query}") 

158 

159 results = [] 

160 for hit in hits: 

161 dish = hit.dish 

162 logger.debug(f"Retrieved dish from hit: {dish}") 

163 results.append(DishData( 

164 dish_id= dish["_id"], 

165 dish_name=dish["name"], 

166 description=dish["description"], 

167 price=dish["price"], 

168 ingredients=dish["ingredients"], 

169 serving_size=dish.get("serving_size"), 

170 availability=dish.get("availability", True), 

171 allergens=[a["allergen"] for a in dish.get("explicit_allergens", [])], 

172 nutrition_facts=dish.get("nutrition_facts", {}) 

173 )) 

174 logging.debug(f"Food item query results: {results}") 

175 return results

178def get_dish_info(state): 

179 """ 

180 Extracts and summarizes detailed dish information for one or more user queries. 

181 

182 This function orchestrates a multi-step retrieval and reasoning pipeline: 

183 1. **Intent Derivation:** Determines whether the query is food-related, factual, 

184 or general knowledge. 

185 2. **Data Retrieval:** Fetches dishes from the target restaurant based on the query. 

186 3. **Filtering & Validation:** Applies user- or query-based filters (e.g., allergens, price) 

187 and ensures retrieved dishes meet relevance criteria. 

188 4. **LLM Summarization:** Passes dish data and the user query to a language model to 

189 generate a structured JSON response. 

190 

191 Args: 

192 state: An object containing: 

193 - `restaurant_id` (str): The target restaurant to query. 

194 - `query_parts` (dict): Parsed user intents grouped by category. 

195 - `current_context` (str, optional): Prior summarized conversation context. 

196 

197 Returns: 

198 dict: A mapping with a single key `"info_results"` containing a `DishInfoResult` 

199 object, which includes structured results per query: 

200 { 

201 "info_results": { 

202 "What is in the pasta?": DishInfoResponse( 

203 dish_name="Penne Alfredo", 

204 requested_info="It includes pasta, cream, garlic, and cheese.", 

205 source_data=[...] 

206 ) 

207 } 

208 } 

209 

210 Raises: 

211 Logs errors internally and continues for each query. Does not raise exceptions outward. 

212 """ 

213 results = {} 

214 restaurant_id = state.restaurant_id 

215 for query in state.query_parts.get("dish_info",[]): 

216 logging.debug(f"Getting dish info for query: {query} and restaurant_id: {restaurant_id}") 

217 

218 if state.current_context: 

219 logging.debug(f"Appending current context to query: {state.current_context}") 

220 query = f"{query}\n\nAdditional context:\n{state.current_context}" 

221 

222 try: 

223 intent = derive_dish_info_intent(query) 

224 except GenericException as e: 

225 results[query] = DishInfoResponse( 

226 dish_name=None, 

227 requested_info="Intent derivation failed", 

228 source_data=[] 

229 ) 

230 continue 

231 

232 logging.debug(f"Query intent: {intent}") 

233 if intent.type == "general_knowledge": 

234 response = handle_general_knowledge(query=query) 

235 results[query] = DishInfoResponse( 

236 dish_name=None, 

237 requested_info=response.answer, 

238 source_data=[] 

239 ) 

240 continue 

241 

242 try: 

243 dishes = handle_food_item_query(query, restaurant_id=restaurant_id) 

244 dishes = apply_filters(query,dishes) 

245 dishes = validate_retrieved_dishes(query,dishes) 

246 except NotFoundException as e: 

247 logger.error(str(e)) 

248 results[query] = DishInfoResponse( 

249 dish_name=None, 

250 requested_info=str(e), 

251 source_data=[] 

252 ) 

253 continue 

254 except Exception as e: 

255 logger.error(str(e)) 

256 results[query] = DishInfoResponse( 

257 dish_name=None, 

258 requested_info=f"Unexpected error: {str(e)}", 

259 source_data=[] 

260 ) 

261 continue 

262 

263 context = "" 

264 context = "\n".join([ 

265 f"Dish Name : {d.dish_name}\n" 

266 f"Description : {d.description}\n" 

267 f"Price : {d.price}\n" 

268 f"Ingredients : {', '.join(d.ingredients or [])}\n" 

269 f"Serving Size: {d.serving_size or 'N/A'}\n" 

270 f"Availability: {d.availability}\n" 

271 f"Allergens : {', '.join(d.allergens or [])}\n" 

272 f"Nutrition : {d.nutrition_facts}\n" 

273 for d in dishes 

274 ]) 

275 

276 prompt = ChatPromptTemplate.from_template(""" 

277 You are a food information assistant. 

278 Using ONLY the following dish data, answer the user's query. 

279 Format the response as JSON: 

280 - "dish_name" - name of the dish being referred to. 

281 - "requested_info" - string answer to the user's query. 

282 - "source_data" - list of relevant dish data used to answer the query. 

283 

284 User Query: {query} 

285 

286 Dish Data: {context} 

287 """) 

288 response = llm.invoke(prompt.format_messages(query=query,context=context)) 

289 logging.debug(f"LLM Response: {response.content}") 

290 

291 try: 

292 # Check if response is empty 

293 if not response.content or not response.content.strip(): 

294 logger.warning(f"Empty LLM response for dish info query: {query}") 

295 results[query] = DishInfoResponse( 

296 dish_name=None, 

297 requested_info="No response generated", 

298 source_data=[] 

299 ) 

300 continue 

301 

302 # Try to extract JSON from markdown code blocks if present 

303 content = response.content.strip() 

304 if content.startswith("```"): 

305 content = content.replace("```json", "").replace("```", "").strip() 

306 

307 response_json = json.loads(content) 

308 results[query] = DishInfoResponse(**response_json) 

309 except json.JSONDecodeError as e: 

310 logger.error(f"JSON decode error in get_dish_info. Response: {response.content[:500]}. Error: {str(e)}") 

311 results[query] = DishInfoResponse( 

312 dish_name=None, 

313 requested_info="Could not parse LLM Response", 

314 source_data=[] 

315 ) 

316 except Exception as e: 

317 logger.error(f"Error parsing dish info response: {str(e)}") 

318 results[query] = DishInfoResponse( 

319 dish_name=None, 

320 requested_info="Could not parse LLM Response", 

321 source_data=[] 

322 ) 

323 

324 # return {"info_results":results} 

325 logger.debug(f"Printing Info Results: {results}") 

326 return {"info_results":DishInfoResult(info_results=results)}
