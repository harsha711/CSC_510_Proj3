1""" 

2Defines Pydantic models for AI-Powered Meal Compatibility Scoring. 

3 

4This module provides structured schemas for multi-factor compatibility analysis 

5between dishes and user dietary profiles, including allergen safety, nutrition matching, 

6taste preferences, and dietary pattern alignment. 

7""" 

8from pydantic import BaseModel, Field 

9from typing import List, Optional, Dict 

10from enum import Enum 

11 

12class SafetyLevel(str, Enum): 

13 """Safety assessment levels for allergen analysis""" 

14 SAFE = "SAFE" 

15 WARNING = "WARNING" 

16 UNSAFE = "UNSAFE" 

17 

18class MatchLevel(str, Enum): 

19 """Match quality levels for compatibility factors""" 

20 EXCELLENT = "EXCELLENT" 

21 GOOD = "GOOD" 

22 MODERATE = "MODERATE" 

23 POOR = "POOR" 

24 

25class AllergenSafetyScore(BaseModel): 

26 """ 

27 Allergen safety analysis for a dish. 

28 

29 Attributes: 

30 score (int): Safety score 0-100 (100 = completely safe). 

31 level (SafetyLevel): Safety assessment (SAFE/WARNING/UNSAFE). 

32 detected_allergens (List[str]): Allergens found that match user allergies. 

33 reasoning (str): Explanation of safety assessment. 

34 """ 

35 score: int = Field(..., ge=0, le=100) 

36 level: SafetyLevel 

37 detected_allergens: List[str] = Field(default_factory=list) 

38 reasoning: str

40class NutritionMatchScore(BaseModel):

42 Nutrition matching analysis based on user health goals.

46 level (MatchLevel): Match quality assessment. 

47 matched_goals (List[str]): Health goals that align with dish nutrition. 

48 conflicts (List[str]): Health goals that conflict with dish nutrition. 

49 reasoning (str): Detailed explanation of nutrition analysis. 

50 """ 

51 score: int = Field(..., ge=0, le=100) 

52 level: MatchLevel

54 conflicts: List[str] = Field(default_factory=list)

106class CompatibilityScore(BaseModel): 

107 """ 

108 Complete compatibility analysis for a dish. 

109 

110 Attributes: 

111 dish_id (str): Unique identifier of the dish. 

112 dish_name (str): Name of the dish. 

113 overall_score (int): Overall compatibility score 0-100.

115 nutrition_match (NutritionMatchScore): Nutrition matching analysis. 

116 taste_preference (TastePreferenceScore): Taste preference analysis.

121 dish_id: str 

122 dish_name: str 

123 overall_score: int = Field(..., ge=0, le=100)
